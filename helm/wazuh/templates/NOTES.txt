MULTI-TENANT - Para adicionar um novo cliente:
  helm install wazuh-cliente2 ./helm/wazuh \
    -n cliente2 --create-namespace \
    -f helm/wazuh/values-production.yaml \
    --set secrets.indexer.password=SenhaNova \
    --set secrets.clusterKey=ChaveUnicaPorCliente123456789012

UPGRADE:
  helm upgrade {{ .Release.Name }} ./helm/wazuh -n {{ .Release.Namespace }} -f values-producao.yaml

DESINSTALAÇÃO:
  helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}
  # ATENÇÃO: Os Secrets de certificados NÃO são deletados (proteção de dados)
  # Para remover: kubectl delete secret -l app.kubernetes.io/instance={{ .Release.Name }} -n {{ .Release.Namespace }}
